#### 5.2 Isi file `db.js`

Buat `db.js` dan isi:

```js
const mysql = require("mysql2/promise");

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: 10,
  queueLimit: 0,
});

module.exports = pool;
```

#### 5.3 Isi file `server.js`

Buat `server.js` dan isi:

```js
require("dotenv").config();
const express = require("express");
const cors = require("cors");
const pool = require("./db");

const app = express();

app.use(cors());
app.use(express.json());

// health check
app.get("/health", async (req, res) => {
  res.json({ status: "ok", message: "API berjalan" });
});

// db check
app.get("/db-check", async (req, res) => {
  try {
    const [rows] = await pool.query("SELECT 1 AS test");
    res.json({ status: "ok", db: rows[0] });
  } catch (err) {
    res.status(500).json({ status: "error", message: err.message });
  }
});

app.listen(process.env.PORT, () => {
  console.log(`Server jalan di http://localhost:${process.env.PORT}`);
});
```

```js
app.post("/assets", async (req, res) => {
  try {
    const {
      asset_code,
      asset_name,
      asset_type = "Other",
      location = null,
      status = "Active",
      installed_date = null,
      notes = null,
    } = req.body;

    // Validasi sederhana
    if (!asset_code || !asset_name) {
      return res.status(400).json({
        status: "error",
        message: "asset_code dan asset_name wajib diisi",
      });
    }

    const [result] = await pool.query(
      `INSERT INTO asset
       (asset_code, asset_name, asset_type, location, status, installed_date, notes)
       VALUES (?, ?, ?, ?, ?, ?, ?)`,
      [asset_code, asset_name, asset_type, location, status, installed_date, notes]
    );

    res.status(201).json({
      status: "ok",
      message: "Asset berhasil ditambahkan",
      asset_id: result.insertId,
    });
  } catch (err) {
    res.status(500).json({ status: "error", message: err.message });
  }
});
```

## 1) Tambahkan 3 endpoint CRUD lanjutan (copy paste)

```js
/* GET /assets/:id: detail asset */
app.get("/assets/:id", async (req, res) => {
  try {
    const id = Number(req.params.id);

    const [rows] = await pool.query(
      `SELECT asset_id, asset_code, asset_name, asset_type, location, status,
              installed_date, notes, created_at, updated_at
       FROM asset
       WHERE asset_id = ?`,
      [id]
    );

    if (rows.length === 0) {
      return res
        .status(404)
        .json({ status: "error", message: "Asset tidak ditemukan" });
    }

    res.json({ status: "ok", data: rows[0] });
  } catch (err) {
    res.status(500).json({ status: "error", message: err.message });
  }
});

/* PUT /assets/:id: update asset */
app.put("/assets/:id", async (req, res) => {
  try {
    const id = Number(req.params.id);

    const {
      asset_code,
      asset_name,
      asset_type,
      location,
      status,
      installed_date,
      notes,
    } = req.body;

    const [result] = await pool.query(
      `UPDATE asset
       SET asset_code = COALESCE(?, asset_code),
           asset_name = COALESCE(?, asset_name),
           asset_type = COALESCE(?, asset_type),
           location = COALESCE(?, location),
           status = COALESCE(?, status),
           installed_date = COALESCE(?, installed_date),
           notes = COALESCE(?, notes)
       WHERE asset_id = ?`,
      [
        asset_code,
        asset_name,
        asset_type,
        location,
        status,
        installed_date,
        notes,
        id,
      ]
    );

    if (result.affectedRows === 0) {
      return res
        .status(404)
        .json({ status: "error", message: "Asset tidak ditemukan" });
    }

    res.json({ status: "ok", message: "Asset berhasil diupdate" });
  } catch (err) {
    res.status(500).json({ status: "error", message: err.message });
  }
});

/* DELETE /assets/:id: hapus asset */
app.delete("/assets/:id", async (req, res) => {
  try {
    const id = Number(req.params.id);

    const [result] = await pool.query(`DELETE FROM asset WHERE asset_id = ?`, [
      id,
    ]);

    if (result.affectedRows === 0) {
      return res
        .status(404)
        .json({ status: "error", message: "Asset tidak ditemukan" });
    }

    res.json({ status: "ok", message: "Asset berhasil dihapus" });
  } catch (err) {
    res.status(500).json({ status: "error", message: err.message });
  }
});
```

## 4) Buat halaman daftar asset

Buat file:

**`frontend/src/components/AssetTable.vue`**

```vue
<template>
  <div class="wrap">
    <h1>Daftar Asset Data Center</h1>

    <div class="toolbar">
      <button @click="fetchAssets" :disabled="loading">
        {{ loading ? "Loading..." : "Refresh" }}
      </button>
      <span v-if="error" class="error">{{ error }}</span>
    </div>

    <table v-if="assets.length" class="table">
      <thead>
        <tr>
          <th>ID</th>
          <th>Code</th>
          <th>Nama</th>
          <th>Tipe</th>
          <th>Lokasi</th>
          <th>Status</th>
          <th>Installed</th>
        </tr>
      </thead>
      <tbody>
        <tr v-for="a in assets" :key="a.asset_id">
          <td>{{ a.asset_id }}</td>
          <td>{{ a.asset_code }}</td>
          <td>{{ a.asset_name }}</td>
          <td>{{ a.asset_type }}</td>
          <td>{{ a.location ?? "-" }}</td>
          <td>{{ a.status }}</td>
          <td>{{ a.installed_date ?? "-" }}</td>
        </tr>
      </tbody>
    </table>

    <p v-else-if="!loading && !error">Data asset masih kosong.</p>
  </div>
</template>

<script setup>
import { onMounted, ref } from "vue";
import { api } from "../api";

const assets = ref([]);
const loading = ref(false);
const error = ref("");

async function fetchAssets() {
  loading.value = true;
  error.value = "";
  try {
    const res = await api.get("/assets");
    assets.value = res.data?.data ?? [];
  } catch (err) {
    error.value = err?.response?.data?.message || err.message || "Gagal ambil data";
  } finally {
    loading.value = false;
  }
}

onMounted(fetchAssets);
</script>

<style scoped>
.wrap { max-width: 1000px; margin: 24px auto; font-family: Arial, sans-serif; }
.toolbar { display: flex; gap: 12px; align-items: center; margin-bottom: 12px; }
.error { color: #b00020; }
.table { width: 100%; border-collapse: collapse; }
.table th, .table td { border: 1px solid #ddd; padding: 8px; }
.table th { background: #f4f4f4; text-align: left; }
button { padding: 8px 12px; cursor: pointer; }
</style>
```

```js
function startEdit(a) {
  editingId.value = a.asset_id;
  editForm.value = {
    asset_code: a.asset_code ?? "",
    asset_name: a.asset_name ?? "",
    asset_type: a.asset_type ?? "Other",
    location: a.location ?? "",
    status: a.status ?? "Active",
    installed_date: a.installed_date ?? "",
    notes: a.notes ?? "",
  };
}

function cancelEdit() {
  editingId.value = null;
}

async function saveEdit(id) {
  loading.value = true;
  error.value = "";
  try {
    await api.put(`/assets/${id}`, {
      asset_code: editForm.value.asset_code,
      asset_name: editForm.value.asset_name,
      asset_type: editForm.value.asset_type,
      location: editForm.value.location || null,
      status: editForm.value.status,
      installed_date: editForm.value.installed_date || null,
      notes: editForm.value.notes || null,
    });
    editingId.value = null;
    await fetchAssets();
  } catch (err) {
    error.value = err?.response?.data?.message || err.message || "Gagal update data";
  } finally {
    loading.value = false;
  }
}

async function remove(id) {
  const ok = confirm("Yakin hapus asset ini?");
  if (!ok) return;

  loading.value = true;
  error.value = "";
  try {
    await api.delete(`/assets/${id}`);
    await fetchAssets();
  } catch (err) {
    error.value = err?.response?.data?.message || err.message || "Gagal hapus data";
  } finally {
    loading.value = false;
  }
}
```

## 3) Buat router (src/router.js)

```js
import { createRouter, createWebHistory } from "vue-router";

import Login from "./pages/Login.vue";
import Dashboard from "./pages/Dashboard.vue";
import Events from "./pages/Events.vue";
import Assets from "./pages/Assets.vue";

// auth sederhana (sementara) -> simpan token palsu
function isAuthed() {
  return !!localStorage.getItem("token");
}

const router = createRouter({
  history: createWebHistory(),
  routes: [
    { path: "/", redirect: "/dashboard" },
    { path: "/login", component: Login },
    { path: "/dashboard", component: Dashboard, meta: { requiresAuth: true } },
    { path: "/events", component: Events, meta: { requiresAuth: true } },
    { path: "/assets", component: Assets, meta: { requiresAuth: true } },
  ],
});

// proteksi halaman (kalau belum login -> lempar ke /login)
router.beforeEach((to) => {
  if (to.meta.requiresAuth && !isAuthed()) return "/login";
  if (to.path === "/login" && isAuthed()) return "/dashboard";
  return true;
});

export default router;
```

# Penjelasan pelan-pelan (yang sekarang Anda pakai)

## 1️⃣ Login Anda sekarang pakai apa?

Login Anda sekarang pakai:

✅ **Frontend-only authentication (dummy token)**
❌ **TIDAK pakai database user**
❌ **TIDAK pakai backend login**

### Mekanismenya:

Di `Login.vue` kita punya kode ini:

```js
localStorage.setItem("token", "dummy-token");
```

Artinya:

* Begitu user klik Login
* Kita **menyimpan string palsu** `"dummy-token"` di browser
* Browser menganggap: “user sudah login”

---

## 2️⃣ Kenapa setelah login bisa masuk Dashboard?

Karena di `router.js` ada kode ini:

```js
function isAuthed() {
  return !!localStorage.getItem("token");
}
```

Artinya:

* Kalau `localStorage` **punya `token` → dianggap login**
* Kalau tidak ada → dianggap belum login

Dan ini dijaga oleh **route guard**:

```js
router.beforeEach((to) => {
  if (to.meta.requiresAuth && !isAuthed()) return "/login";
});
```

👉 Jadi:

* `/dashboard`, `/assets`, `/events` **tidak bisa diakses**
* kecuali `localStorage.token` ada

---

## 3️⃣ Logout pakai apa?

Di `AppLayout.vue`:

```js
localStorage.removeItem("token");
```

Artinya:

* Token dihapus
* Router mendeteksi user **tidak login**
* User dikirim ke `/login`

---

## 4️⃣ Kenapa ini TIDAK AMAN?

Karena:

* Siapa pun bisa buka DevTools
* Ketik:

  ```js
  localStorage.setItem("token", "apa-aja")
  ```
* Langsung dianggap login 😅

📌 **Ini sengaja kita lakukan di awal** supaya:

* Anda paham **alur aplikasi**
* Anda paham **routing, layout, halaman**
* Tidak langsung pusing ke JWT & crypto

# Sekarang yang BENAR secara industri (yang akan kita buat)

## Login yang BENAR = JWT Authentication

Nanti kita akan pakai:

### Backend

* Tabel `users`
* Password di-**hash** (bcrypt)
* Endpoint:

  * `POST /auth/login`
  * `POST /auth/register`
* Server mengeluarkan **JWT token**

### Frontend

* Login → kirim email & password ke backend
* Backend balas:

  ```json
  { "token": "JWT-ASLI" }
  ```
* Token disimpan di:

  * `localStorage` (belajar)
  * atau `httpOnly cookie` (lebih aman)

---

## Alur login yang BENAR (gambarannya)

```
User isi email + password
        ↓
Frontend kirim ke backend
        ↓
Backend cek user di DB
        ↓
Password cocok?
        ↓
Backend buat JWT
        ↓
Frontend simpan JWT
        ↓
JWT dikirim di setiap request API
```

Contoh request API nanti:

```http
GET /assets
Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...
```
